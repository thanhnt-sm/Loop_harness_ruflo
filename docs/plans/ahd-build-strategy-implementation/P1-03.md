### 4.2 context_budget.py (New Module)

```python
"""Context Budget Manager + Progressive Tool Selector.

Calculates context usage, selects appropriate tools per task,
enforces budget limits.
"""

from __future__ import annotations
from dataclasses import dataclass
from typing import Any

# Context window sizes per model (approximate)
MODEL_CONTEXT_WINDOWS = {
    "default": 8192,
    "glm-5.2": 200000,
    "kimi-k2.7": 128000,
    "lightning": 200000,
    "small": 8192,
}

DEFAULT_CONTEXT_WINDOW = 8192


@dataclass
class ContextBudget:
    """Context budget tracking for a task."""
    model: str = "default"
    window_size: int = DEFAULT_CONTEXT_WINDOW
    used_tokens: int = 0
    reserved_tokens: int = 0  # system prompt, pinned memory

    @property
    def available_tokens(self) -> int:
        return max(0, self.window_size - self.reserved_tokens - self.used_tokens)

    @property
    def usage_pct(self) -> float:
        if self.window_size == 0:
            return 0.0
        return (self.used_tokens + self.reserved_tokens) / self.window_size * 100

    def is_alert_threshold(self) -> bool:
        return self.usage_pct >= CONTEXT_BUDGET_ALERT_PCT

    def is_kill_threshold(self) -> bool:
        return self.usage_pct >= CONTEXT_BUDGET_KILL_PCT


def estimate_tokens(text: str) -> int:
    """Rough token estimation: ~4 chars per token."""
    return max(1, len(text) // 4)


def select_progressive_tools(
    task_type: str,
    available_tools: list[str],
    limit: int = PROGRESSIVE_TOOL_LIMIT,
) -> list[str]:
    """Select 3-4 context-appropriate tools for task type.

    Task types: 'read', 'write', 'edit', 'search', 'bash', 'mcp', 'general'
    """
    # Tool categories by task type
    TOOL_CATEGORIES = {
        "read": ["Read", "Glob", "Grep", "LS"],
        "write": ["Write", "Edit", "MultiEdit", "Bash"],
        "edit": ["Edit", "MultiEdit", "Read", "Grep"],
        "search": ["Grep", "Glob", "Read", "LS"],
        "bash": ["Bash", "Read", "Write", "Grep"],
        "mcp": ["mcp__*", "Read", "Write"],
        "general": ["Read", "Write", "Bash", "Grep"],
    }

    category_tools = TOOL_CATEGORIES.get(task_type, TOOL_CATEGORIES["general"])

    # Filter available tools matching category
    selected = []
    for pattern in category_tools:
        if pattern.endswith("*"):
            # Wildcard for MCP tools
            prefix = pattern[:-1]
            matches = [t for t in available_tools if t.startswith(prefix)]
            selected.extend(matches[:limit - len(selected)])
        else:
            if pattern in available_tools:
                selected.append(pattern)
        if len(selected) >= limit:
            break

    return selected[:limit]
```
